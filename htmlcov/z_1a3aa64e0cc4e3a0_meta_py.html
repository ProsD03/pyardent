1"""Read-only models for the API's metadata/statistics endpoints. 

2 

3Unlike the resource models in the rest of `models/`, these are flat DTOs: 

4they don't carry a `_client` reference or expose further `get_*` traversal 

5methods, since the endpoints they map to (`/version`, `/stats`, 

6`/stats/stations/*`) are dead ends, not nodes in the API's resource graph. 

7""" 

8 

9from datetime import datetime 

10 

11from pydantic import BaseModel, ConfigDict, Field 

12from pydantic.alias_generators import to_camel, to_pascal 

13
